\newglossaryentry{topic}{
    name={Topic},
    description={The highest level of semantic abstraction in this study, representing the broad "layers of the onion" in Social Penetration Theory. 
    \\\textbf{Synonyms:} Categories, Macro-dimensions, Main Themes, SPT Layers.}
}

\newglossaryentry{subtopic}{
    name={Subtopic},
    description={The fundamental and granular unit of self-disclosure analysis. These are specific themes (e.g., mental health, family dynamics) that provide the raw data for scoring. 
    \\\textbf{Synonyms:} Sub-dimensions, Sub-themes, Granular Features.}
}


\newglossaryentry{inference}{
    name={Inference},
    description={The process of executing a trained model (in this case, Gemma-2-9b-it) to generate predictions or evaluations on new, unseen data based on the provided input or prompt.}
}

\newglossaryentry{GDPR}{

    description={General Data Protection Regulation; the legal framework for data privacy and protection in the European Union \\\textbf{Synonyms:} RGPD }

\newglossaryentry{spt}{
    name={Social Penetration Theory},
    description={The fundamental theoretical framework used to understand how individuals reveal personal information to others. It uses the "onion metaphor" to suggest that personality is structured in various layers (peripheral, intermediate, and core), and that the development of a relationship involves systematically peeling back these layers through communication}
}

\newglossaryentry{hadamard}{
    name={Hadamard Product},
    description={A mathematical operation applied to integrate subtopics into toipcs. It functions as a filtering mechanism that multiplies vectors element-wise to eliminate low-confidence model assignments while preserving the intensity of high-quality self-disclosures}
}
